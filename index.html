const SPREADSHEET_ID = "1Pnpm20XBN8SHWoaGT7deblrMlcUBg6QMUi4j3EifTEQ"; const SHEET_CHORISTES = "Choristes"; const SHEET_PRESENCES = "Présences"; const NOM_CHORALE = "Suis ta voix"; const NOM_SCANNER = "Scanner 1"; function doGet(e) { const id = e.parameter.id; if (!id) { return ContentService .createTextOutput("ID manquant"); } enregistrerPresence(id); return ContentService .createTextOutput("OK"); } function enregistrerPresence(id) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const choristesSheet = ss.getSheetByName(SHEET_CHORISTES); const presencesSheet = ss.getSheetByName(SHEET_PRESENCES); if (!choristesSheet || !presencesSheet) { throw new Error( "Onglet Choristes ou Présences introuvable" ); } /* ========================= RECHERCHE DU CHORISTE ========================= */ const data = choristesSheet .getDataRange() .getValues(); let choriste = null; for (let i = 1; i < data.length; i++) { if ( String(data[i][0]).trim() === String(id).trim() ) { if ( String(data[i][4]) .trim() .toUpperCase() !== "OUI" ) { ajouterLigne( presencesSheet, id, "", "", "", "INACTIF" ); return; } choriste = { id: data[i][0], prenom: data[i][1], nom: data[i][2], voix: data[i][3] }; break; } } /* ========================= QR INCONNU ========================= */ if (!choriste) { ajouterLigne( presencesSheet, id, "", "", "", "QR INCONNU" ); return; } /* ========================= VÉRIFICATION DOUBLON ========================= */ const maintenant = new Date(); const presenceData = presencesSheet .getDataRange() .getValues(); for ( let i = presenceData.length - 1; i >= 1; i-- ) { const dateScan = presenceData[i][0]; const idScan = String(presenceData[i][2]).trim(); if ( idScan === String(id).trim() && dateScan instanceof Date ) { const difference = ( maintenant.getTime() - dateScan.getTime() ) / 1000; if (difference < 120) { ajouterLigne( presencesSheet, choriste.id, choriste.prenom, choriste.nom, choriste.voix, "DÉJÀ ENREGISTRÉ" ); return; } break; } } /* ========================= ENREGISTREMENT NORMAL ========================= */ ajouterLigne( presencesSheet, choriste.id, choriste.prenom, choriste.nom, choriste.voix, "OK" ); } /* ========================= AJOUT D'UNE LIGNE ========================= */ function ajouterLigne( sheet, id, prenom, nom, voix, statut ) { const maintenant = new Date(); sheet.appendRow([ maintenant, // Date maintenant, // Heure id, // ID prenom, // Prénom nom, // Nom voix, // Voix statut, // Statut NOM_CHORALE, // Chorale NOM_SCANNER // Appareil ]); }
